2. 後端 Google Apps Script (`Code.gs`) 完整程式碼： 
/**
 * 社團法人台灣帕金森病友權益促進會 (tpda.org.tw)
 * 官網前台即時 CMS 雲端同步後端 (Google Apps Script)
 * 部署步驟：
 * 1. 開啟一個新的 Google 試算表，點選「擴充功能」>「Apps Script」
 * 2. 貼上本程式碼並儲存，點選右上角「部署」>「新增部署作業」
 * 3. 類型選擇「網頁應用程式」，執行身分選「我」，誰可以存取選「所有人」
 * 4. 複製產生的網頁應用程式 URL，貼回官網後台設定框即可完成同步！
 */
const SHEET_NAME = 'TPDA_CMS_DATA';

function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(['Key', 'JsonData', 'UpdatedAt']);
  }
  return sheet;
}

// 讀取官網最新內容 (GET)
function doGet(e) {
  const sheet = getOrCreateSheet();
  const lastRow = sheet.getLastRow();
  let payload = { status: 'empty', data: null };
  if (lastRow >= 2) {
    const raw = sheet.getRange(2, 2).getValue();
    if (raw) {
      payload = {
        status: 'ok',
        updatedAt: sheet.getRange(2, 3).getValue(),
        data: JSON.parse(raw)
      };
    }
  }
  return ContentService.createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
}

// 寫入官網即時修改內容 (POST)
function doPost(e) {
  try {
    const sheet = getOrCreateSheet();
    const body = JSON.parse(e.postData.contents);
    const now = new Date().toISOString();
    const jsonString = JSON.stringify(body.data);
    
    if (sheet.getLastRow() < 2) {
      sheet.appendRow(['SITE_STATE', jsonString, now]);
    } else {
      sheet.getRange(2, 1, 1, 3).setValues([['SITE_STATE', jsonString, now]]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      updatedAt: now
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
